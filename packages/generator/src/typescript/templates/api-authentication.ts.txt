import { createRemoteJWKSet, errors, jwtVerify, type JWTPayload, type JWTVerifyGetKey } from "jose";

import {
  AUTHENTICATION,
  type Principal,
  principalFromClaims,
  Unauthenticated,
} from "../security.js";
import type { Authenticator } from "./runtime.js";

/**
 * Where the verification key comes from: a JWKS URL (fetched, cached and refreshed by jose), one
 * key (a public `CryptoKey`, or the secret bytes for HS* algorithms), or your own key lookup.
 */
export type JwtKey =
  | { readonly jwksUrl: string | URL }
  | { readonly key: CryptoKey | Uint8Array }
  | { readonly getKey: JWTVerifyGetKey };

export type BearerJwtOptions = JwtKey & {
  /** Expected `iss`; default `security.authentication.issuer`. */
  readonly issuer?: string;
  /** Expected `aud`; default `security.authentication.audience`. */
  readonly audience?: string;
  /** Allowed clock skew in seconds for `exp` / `nbf`; default `authentication.clock_tolerance`. */
  readonly clockTolerance?: number;
};

/**
 * The token of an `Authorization: Bearer <token>` header (RFC 6750 §2.1): undefined without the
 * header, null when the header is not a bearer token.
 */
export function bearerToken(request: Request): string | null | undefined {
  const header = request.headers.get("authorization");
  if (header === null) return undefined;
  const match = /^Bearer +([A-Za-z0-9\-._~+/]+=*) *$/i.exec(header);
  return match?.[1] ?? null;
}

function keyLookup(source: JwtKey): JWTVerifyGetKey {
  if ("jwksUrl" in source) return createRemoteJWKSet(new URL(source.jwksUrl));
  if ("getKey" in source) return source.getKey;
  const key = source.key;
  return () => key;
}

function invalidToken(message: string): Unauthenticated {
  return new Unauthenticated({ error: "invalid_token" }, message);
}

/**
 * Authenticates requests by a bearer JWT (RFC 7519) following the JWT best current practices (RFC
 * 8725): the signature is verified with the configured key and only the model's algorithms
 * (`none` never), `iss` and `aud` must match, `exp` and `sub` must be present, `exp` / `nbf` are
 * checked with a small clock tolerance. The claims become a Principal (`principalFromClaims`).
 *
 * No `Authorization` header gives null (the handler answers 401 `WWW-Authenticate: Bearer`); a bad
 * token throws Unauthenticated with `error: "invalid_token"`. A JWKS that cannot be fetched is a
 * server error (500), not the caller's fault.
 *
 * ```ts
 * const authenticate = createBearerJwtAuthenticator({ jwksUrl: process.env.JWKS_URL });
 * const handler = createApiHandler(dependencies, { authenticate });
 * ```
 */
export function createBearerJwtAuthenticator(options: BearerJwtOptions): Authenticator<Principal> {
  const issuer = options.issuer ?? AUTHENTICATION.issuer;
  const audience = options.audience ?? AUTHENTICATION.audience;
  if (issuer === undefined || audience === undefined) {
    throw new Error("A bearer JWT authenticator needs an issuer and an audience");
  }
  const getKey = keyLookup(options);
  const verifyOptions = {
    issuer,
    audience,
    algorithms: [...AUTHENTICATION.algorithms],
    clockTolerance: options.clockTolerance ?? AUTHENTICATION.clockTolerance,
    requiredClaims: ["exp", "sub"],
  };
  return async (request) => {
    const token = bearerToken(request);
    if (token === undefined) return null;
    if (token === null) throw invalidToken("The Authorization header is not a bearer token");
    let payload: JWTPayload;
    try {
      ({ payload } = await jwtVerify(token, getKey, verifyOptions));
    } catch (error) {
      if (error instanceof errors.JWTExpired) throw invalidToken("The token has expired");
      if (error instanceof errors.JOSEError && !(error instanceof errors.JWKSTimeout)) {
        throw invalidToken("The token is invalid");
      }
      throw error;
    }
    const principal = principalFromClaims(payload);
    if (principal === null) throw invalidToken("The token does not describe a principal");
    return principal;
  };
}
