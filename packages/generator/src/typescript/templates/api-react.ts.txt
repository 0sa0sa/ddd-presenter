import { createContext, useContext } from "react";

import type { ApiClient } from "./client.js";

/**
 * The API client the generated hooks use. Provide it once, next to the QueryClientProvider:
 * `<ApiClientContext value={createApiClient({ baseUrl })}>` (React 19) or
 * `<ApiClientContext.Provider value={…}>` (React 18).
 */
export const ApiClientContext = createContext<ApiClient | null>(null);

/** The API client of the nearest ApiClientContext (throws when none is provided). */
export function useApiClient(): ApiClient {
  const api = useContext(ApiClientContext);
  if (api === null) throw new Error("useApiClient: wrap the app in <ApiClientContext value={api}>");
  return api;
}
