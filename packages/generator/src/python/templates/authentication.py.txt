from __future__ import annotations

import re

import jwt

from __PKG__.generated.security import (
    JWT_ALGORITHMS,
    JWT_AUDIENCE,
    JWT_ISSUER,
    JWT_LEEWAY,
    Principal,
    Unauthenticated,
    principal_from_claims,
)

__all__ = ["BearerJwtAuthenticator", "bearer_token"]

_BEARER = re.compile(r"^Bearer +([A-Za-z0-9\-._~+/]+=*) *$", re.IGNORECASE)


def _invalid_token(message: str) -> Unauthenticated:
    return Unauthenticated(message, error="invalid_token")


def bearer_token(authorization: str | None) -> str | None:
    """The token of an `Authorization: Bearer <token>` header value (RFC 6750 §2.1).

    None without the header; raises Unauthenticated (`error="invalid_token"`) for another scheme.
    """
    if authorization is None:
        return None
    match = _BEARER.match(authorization)
    if match is None:
        raise _invalid_token("The Authorization header is not a bearer token")
    return match.group(1)


class BearerJwtAuthenticator:
    """Authenticates a request by its bearer JWT (RFC 7519) with PyJWT, following RFC 8725.

    The signature is verified with the configured key and only the model's algorithms (`none`
    never), `iss` and `aud` must match, `exp` and `sub` must be present, `exp` / `nbf` are checked
    with a small leeway. The claims become a Principal (`principal_from_claims`).

    Framework-agnostic: pass the Authorization header value. With FastAPI::

        authenticator = BearerJwtAuthenticator(jwks_url="https://issuer/.well-known/jwks.json")

        def principal(authorization: str | None = Header(default=None)) -> Principal | None:
            return authenticator.authenticate(authorization)

    and answer Unauthenticated with 401 and `WWW-Authenticate: Bearer` (adding
    `error="invalid_token"` when `error.details["error"]` says so), NotAuthorized with 403.
    """

    def __init__(
        self,
        *,
        key: str | bytes | jwt.PyJWK | None = None,
        jwks_url: str | None = None,
        issuer: str | None = None,
        audience: str | None = None,
        leeway: int = JWT_LEEWAY,
    ) -> None:
        if (key is None) == (jwks_url is None):
            raise ValueError("Pass exactly one of key or jwks_url")
        resolved_issuer = issuer if issuer is not None else JWT_ISSUER
        resolved_audience = audience if audience is not None else JWT_AUDIENCE
        if resolved_issuer is None or resolved_audience is None:
            raise ValueError("A bearer JWT authenticator needs an issuer and an audience")
        self._key = key
        self._jwks = jwt.PyJWKClient(jwks_url) if jwks_url is not None else None
        self._issuer = resolved_issuer
        self._audience = resolved_audience
        self._leeway = leeway

    def authenticate(self, authorization: str | None) -> Principal | None:
        """The principal of an Authorization header value; None without one.

        Raises Unauthenticated with `error="invalid_token"` for a bad token. A JWKS that cannot be
        fetched raises PyJWKClientConnectionError (a server error, not the caller's fault).
        """
        token = bearer_token(authorization)
        if token is None:
            return None
        try:
            key = self._key if self._jwks is None else self._jwks.get_signing_key_from_jwt(token)
            claims = jwt.decode(
                token,
                key if key is not None else "",
                algorithms=list(JWT_ALGORITHMS),
                audience=self._audience,
                issuer=self._issuer,
                leeway=self._leeway,
                options={"require": ["exp", "sub", "iss", "aud"]},
            )
        except jwt.PyJWKClientConnectionError:
            raise
        except jwt.ExpiredSignatureError as exc:
            raise _invalid_token("The token has expired") from exc
        except jwt.PyJWTError as exc:
            raise _invalid_token("The token is invalid") from exc
        principal = principal_from_claims(claims)
        if principal is None:
            raise _invalid_token("The token does not describe a principal")
        return principal
